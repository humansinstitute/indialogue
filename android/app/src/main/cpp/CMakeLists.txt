cmake_minimum_required(VERSION 3.22.1)
project(dialogue LANGUAGES C CXX)
include(FetchContent)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(LLAMA_CURL OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_CPU_KLEIDIAI OFF CACHE BOOL "" FORCE)
FetchContent_Declare(llama URL https://codeload.github.com/ggml-org/llama.cpp/tar.gz/a7a98e0fffed794396b3fbad4dcdbbc184963645 URL_HASH SHA256=4d78e6aa4a9124b58dff994416525294fbab2990fe905a640d4cbd26bf563a31)
FetchContent_MakeAvailable(llama)
add_library(dialogue SHARED cleanup.cpp)
target_link_libraries(dialogue llama android log)
target_compile_features(dialogue PRIVATE cxx_std_17)
